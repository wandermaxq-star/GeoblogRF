        </svg>
      ),
      path: '/pro',
      color: '#9C27B0',
    },
  ];

  const toggleMenu = () => {
    setIsMenuOpen(!isMenuOpen);
  };

  const handleNavigation = (path: string) => {
    navigate(path);
    setIsMenuOpen(false);
  };

  const isActive = (path: string) => location.pathname === path;

  return (
    <div className="bottom-navigation">
      {/* Раскрывающееся меню */}
      <div className={`speed-dial ${isMenuOpen ? 'open' : ''}`}>
        {speedDialItems.map((item, index) => {
          const angle = ((index - speedDialItems.length / 2 + 0.5) * 30) * (Math.PI / 180);
          const radius = isMenuOpen ? 100 : 0;
          const x = Math.sin(angle) * radius;
          const y = -Math.cos(angle) * radius - 20;

          return (
            <button
              key={item.id}
              className="speed-dial-item"
              style={{
                transform: `translate(${x}px, ${y}px)`,
                opacity: isMenuOpen ? 1 : 0,
                transitionDelay: `${index * 50}ms`,
              }}
              onClick={() => handleNavigation(item.path)}
            >
              <div className="speed-dial-icon" style={{ backgroundColor: item.color || '#333' }}>
                {item.icon}
              </div>
              <span className="speed-dial-label">{item.label}</span>
            </button>
          );
        })}
      </div>

      {/* Основная навигация */}
      <div className="nav-bar">
        {mainNavItems.map((item) => (
          <button
            key={item.id}
            className={`nav-item ${isActive(item.path) ? 'active' : ''}`}
            onClick={() => handleNavigation(item.path)}
          >
            <div className="nav-icon">{item.icon}</div>
            <span className="nav-label">{item.label}</span>
          </button>
        ))}

        {/* Центральная кнопка + */}
        <button
          className={`nav-item center-button ${isMenuOpen ? 'open' : ''}`}
          onClick={toggleMenu}
        >
          <div className="plus-icon">
            <span className="plus-line plus-vertical"></span>
            <span className="plus-line plus-horizontal"></span>
          </div>
        </button>
      </div>

      {/* Затемнение фона при открытом меню */}
      {isMenuOpen && (
        <div className="overlay" onClick={toggleMenu}></div>
      )}
    </div>
  );
};

export default BottomNavigation;
