  <div className="p-4 border-b flex items-center gap-3 glass-l1">
    <button 
      onClick={() => setMode('select')}
      className="p-2 rounded-full hover:bg-white/20 transition-colors"
    >
      <RotateCcw size={20} />
    </button>
    <h2 className="text-xl font-semibold">
      {mode === 'instant' && 'Момент здесь и сейчас'}
      {mode === 'story' && 'Впечатление дня'}
      {mode === 'guide' && 'Мой путеводитель'}
      {mode === 'continue' && 'Продолжение черновика'}
    </h2>
  </div>
);

// Общий блок эмоционального стартера + основной текст + фото
const renderCommonEditorPart = () => (
  <div className="space-y-6">
    {/* Эмоциональные стартеры */}
    <div className="flex flex-wrap gap-2">
      {[
        "Это место выбило меня из колеи потому что…",
        "Если у тебя всего один день — делай так",
        "Никому не рассказывайте, но здесь…",
        "Самое вкусное/красивое/странное, что я нашёл"
      ].map((starter) => (
        <button
          key={starter}
          onClick={() => setBody(prev => prev ? `${prev}\n\n${starter}` : starter)}
          className="text-sm px-4 py-2 rounded-full bg-blue-50/80 dark:bg-blue-900/30 
                     hover:bg-blue-100 dark:hover:bg-blue-800/40 text-blue-700 dark:text-blue-300
                     border border-blue-200/50 dark:border-blue-700/50 transition-all"
        >
          {starter}
        </button>
      ))}
    </div>

    {/* Основное поле текста */}
    <textarea
      value={body}
      onChange={(e) => setBody(e.target.value)}
      placeholder={
        mode === 'instant' ? "Что происходит прямо сейчас? (2–3 предложения хватит)" :
        mode === 'story' ? "Расскажи свои эмоции, впечатления, что запомнилось..." :
        "Вступление к путеводителю или главная мысль..."
      }
      className="w-full min-h-[140px] p-4 rounded-2xl bg-white/30 dark:bg-gray-800/40 
                 backdrop-blur-xl border border-white/20 dark:border-gray-700/50 
                 focus:border-blue-400 focus:ring-2 focus:ring-blue-200/50 resize-none"
    />

    {/* Фото-блок — акцентный */}
    <div className="space-y-3">
      <label className="block text-sm font-medium text-gray-700 dark:text-gray-300">
        {mode === 'instant' ? 'Главный кадр момента' : 'Фотографии (перетаскивай для порядка)'}
      </label>
      
      <div className={`
        aspect-video rounded-2xl overflow-hidden relative cursor-pointer
        border-2 border-dashed ${photoPreviewUrls.length ? 'border-blue-400/50' : 'border-gray-400/50'}
        hover:border-blue-500/70 transition-colors
      `}>
        <input
          type="file"
          accept="image/*"
          multiple
          onChange={handlePhotoSelect}
          className="absolute inset-0 opacity-0 cursor-pointer z-10"
        />
        
        {photoPreviewUrls.length > 0 ? (
          <img 
            src={photoPreviewUrls[0]} 
            alt="Главное фото" 
            className="w-full h-full object-cover"
          />
        ) : (
          <div className="absolute inset-0 flex flex-col items-center justify-center bg-gradient-to-br from-gray-50/80 to-gray-100/80 dark:from-gray-900/60 dark:to-gray-800/60">
            <Camera className="w-12 h-12 text-gray-400 mb-3" />
            <p className="text-lg font-medium text-gray-600 dark:text-gray-300">
              {mode === 'instant' ? 'Сфотографировать сейчас' : 'Добавить фото / галерея'}
            </p>
          </div>
        )}
      </div>

      {/* Мини-превью остальных фото (если больше одного) */}
      {photoPreviewUrls.length > 1 && (
        <div className="grid grid-cols-4 gap-2">
          {photoPreviewUrls.slice(1).map((url, idx) => (
            <div key={idx} className="aspect-square rounded-lg overflow-hidden relative group">
              <img src={url} alt={`Доп. фото ${idx+1}`} className="w-full h-full object-cover" />
              <button 
                onClick={() => handleRemovePhoto(idx + 1)}
                className="absolute top-1 right-1 bg-red-500/80 text-white rounded-full p-1 opacity-0 group-hover:opacity-100 transition-opacity"
              >
                <FaTimes size={12} />
              </button>
            </div>
          ))}
        </div>
      )}
    </div>
  </div>
);

// Рендер в зависимости от режима
return (
  <div className="flex flex-col h-full">
    {renderHeader()}

    <div className="flex-1 overflow-y-auto p-5 sm:p-6 space-y-8">
      {renderCommonEditorPart()}

      {/* Специфические блоки для каждого режима */}
      {mode === 'instant' && (
        <div className="pt-4">
          <p className="text-sm text-gray-500 dark:text-gray-400 mb-3">
            Момент готов? Можно публиковать сразу!
          </p>
          {/* Здесь можно добавить быстрые теги: "закат", "вкусняшка", "вау-вид" */}
        </div>
      )}

      {mode === 'story' && (
        <div className="space-y-6 pt-4 border-t border-gray-200/50 dark:border-gray-700/50">
          <div className="flex items-center gap-3">
            <MapPin className="w-5 h-5 text-blue-500" />
            <span className="font-medium">Привязать место или маршрут</span>
          </div>
          
          {/* Здесь твой выбор hook — метка/маршрут/событие */}
          <button 
            onClick={() => {/* открыть выбор из favorites */}}
            className="w-full py-3 px-4 rounded-xl bg-white/50 dark:bg-gray-800/50 
                       border border-gray-300/50 dark:border-gray-600/50 text-left"
          >
            + Выбрать точку / маршрут / событие
          </button>

          <div className="text-sm text-gray-500 italic">
            Добавь 1–2 предложения про эмоции — это то, что люди запомнят больше всего
          </div>
        </div>
      )}

      {mode === 'guide' && (
        <div className="space-y-6 pt-6 border-t border-gray-200/50 dark:border-gray-700/50">
          <div className="flex items-center justify-between">
            <h3 className="font-semibold text-lg">Секции путеводителя</h3>
            <button 
              onClick={() => {
                setBlocks([...blocks, { id: Date.now(), type: 'section', title: '', content: '' }]);
              }}
              className="flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-xl hover:bg-blue-700"
            >
              <Plus size={16} />
              Добавить секцию
            </button>
          </div>

          {/* Список секций с drag-and-drop (можно подключить react-beautiful-dnd позже) */}
          {blocks.map((block, index) => (
            <motion.div 
              key={block.id}
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              className="p-4 rounded-2xl bg-white/40 dark:bg-gray-800/40 backdrop-blur-xl border border-white/20"
            >
              <div className="flex items-center justify-between mb-3">
                <input
                  value={block.title}
                  onChange={(e) => {
                    const newBlocks = [...blocks];
                    newBlocks[index].title = e.target.value;
                    setBlocks(newBlocks);
                  }}
                  placeholder={`Секция ${index + 1}`}
                  className="font-medium bg-transparent border-b border-gray-300 focus:border-blue-400 outline-none"
                />
                <button 
                  onClick={() => setBlocks(blocks.filter((_, i) => i !== index))}
                  className="text-red-500 hover:text-red-700"
                >
                  <FaTrash size={14} />
                </button>
              </div>
              
              <textarea
                value={block.content}
                onChange={(e) => {
                  const newBlocks = [...blocks];
                  newBlocks[index].content = e.target.value;
                  setBlocks(newBlocks);
                }}
                placeholder="Что здесь важно знать / увидеть / попробовать..."
                className="w-full min-h-[80px] bg-transparent resize-none focus:outline-none"
              />
            </motion.div>
          ))}

          {blocks.length === 0 && (
            <div className="text-center py-8 text-gray-500">
              Добавь первую секцию — например, «Как добраться» или «Где лучше сфотографироваться»
            </div>
          )}
        </div>
      )}

      {mode === 'continue' && (
        <div className="bg-amber-50/50 dark:bg-amber-900/20 p-5 rounded-2xl border border-amber-200/50">
          <h3 className="font-medium mb-3 flex items-center gap-2">
            <RotateCcw size={18} className="text-amber-600" />
            Продолжаем черновик от {new Date().toLocaleDateString('ru-RU')}
          </h3>
          <p className="text-sm text-amber-800/80 dark:text-amber-200/80">
            Здесь можно подгрузить текст/фото из последнего черновика (реализуй через offlinePostsStorage)
          </p>
        </div>
      )}
    </div>

    {/* Нижняя панель действий — общая для всех */}
    <div className="sticky bottom-0 bg-white/80 dark:bg-gray-900/80 backdrop-blur-2xl border-t border-gray-200/50 p-4">
      <div className="flex gap-3 max-w-3xl mx-auto">
        <button 
          onClick={handleSaveOffline}
          className="flex-1 py-3 px-5 rounded-xl bg-gray-100 dark:bg-gray-800 font-medium flex items-center justify-center gap-2"
        >
          <Cloud size={18} />
          Черновик
        </button>
        
        <button 
          onClick={handleCreatePost}
          disabled={loading || !body.trim() || (mode === 'guide' && blocks.length === 0)}
          className="flex-1 py-3 px-5 rounded-xl bg-gradient-to-r from-blue-600 to-indigo-600 text-white font-medium 
                     disabled:opacity-50 disabled:cursor-not-allowed flex items-center justify-center gap-2 shadow-md"
        >
          {loading ? (
            <div className="w-5 h-5 border-2 border-white border-t-transparent rounded-full animate-spin" />
          ) : (
            <>
              Опубликовать
              <ChevronRight size={18} />
            </>
          )}
        </button>
      </div>
    </div>
  </div>
);
